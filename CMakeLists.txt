cmake_minimum_required(VERSION 3.20)
project(atm_er2_optimizer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/reactor_core.hpp" ER2_CORE_SOURCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/gpu_kernel.cu" ER2_GPU_SOURCE)
set(ER2_KERNEL_SOURCE "${ER2_CORE_SOURCE}\n${ER2_GPU_SOURCE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/reactor_core.hpp src/gpu_kernel.cu)
configure_file(src/embedded_kernel.hpp.in generated/embedded_kernel.hpp @ONLY)
add_executable(atm_er2_optimizer src/main.cpp src/simulator.cpp src/gpu.cpp)
target_include_directories(atm_er2_optimizer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
find_package(Threads REQUIRED)
target_link_libraries(atm_er2_optimizer PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
option(ER2_NATIVE "Optimize CPU code for this machine" OFF)
if(MSVC)
  target_compile_options(atm_er2_optimizer PRIVATE /O2 /W4 /fp:precise)
else()
  target_compile_options(atm_er2_optimizer PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(ER2_NATIVE)
    target_compile_options(atm_er2_optimizer PRIVATE -march=native)
  endif()
endif()
if(MINGW)
  target_link_options(atm_er2_optimizer PRIVATE -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME reference_and_float COMMAND atm_er2_optimizer --backend cpu --self-test --threads 2)
endif()
