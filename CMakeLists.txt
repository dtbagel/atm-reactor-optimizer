cmake_minimum_required(VERSION 3.20)
project(atm_er2_optimizer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/reactor_core.hpp" ER2_CORE_SOURCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/gpu_kernel.cu" ER2_GPU_SOURCE)
set(ER2_KERNEL_SOURCE "${ER2_CORE_SOURCE}\n${ER2_GPU_SOURCE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/reactor_core.hpp src/gpu_kernel.cu)
configure_file(src/embedded_kernel.hpp.in generated/embedded_kernel.hpp @ONLY)
add_library(er2_engine STATIC src/main.cpp src/simulator.cpp src/gpu.cpp)
target_compile_definitions(er2_engine PRIVATE ER2_NO_MAIN)
target_include_directories(er2_engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(er2_engine PUBLIC src)
add_executable(atm_er2_optimizer src/cli.cpp)
find_package(Threads REQUIRED)
target_link_libraries(er2_engine PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_link_libraries(atm_er2_optimizer PRIVATE er2_engine)
option(ER2_NATIVE "Optimize CPU code for this machine" OFF)
if(MSVC)
  target_compile_options(er2_engine PRIVATE /O2 /W4 /fp:precise)
else()
  target_compile_options(er2_engine PRIVATE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(ER2_NATIVE)
    target_compile_options(er2_engine PRIVATE -march=native)
  endif()
endif()
option(ER2_GUI "Build the Windows Dear ImGui / DirectX 11 interface" ON)
if(WIN32 AND ER2_GUI)
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
  add_executable(atm_er2_gui WIN32 src/gui.cpp
    ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(atm_er2_gui PRIVATE ${IMGUI_DIR})
  target_link_libraries(atm_er2_gui PRIVATE er2_engine d3d11 d3dcompiler dwmapi ole32 uuid shell32 windowscodecs gdi32 imm32)
  if(MINGW)
    target_link_options(atm_er2_gui PRIVATE -static -municode)
  endif()
endif()
if(MINGW)
  target_link_options(atm_er2_optimizer PRIVATE -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME reference_and_float COMMAND atm_er2_optimizer --backend cpu --self-test --threads 2)
  add_executable(engine_control_test tests/engine_control.cpp)
  target_link_libraries(engine_control_test PRIVATE er2_engine)
  if(MINGW)
    target_link_options(engine_control_test PRIVATE -static)
  endif()
  add_test(NAME early_finish COMMAND engine_control_test "${CMAKE_CURRENT_BINARY_DIR}/test_results/early_finish")
  add_executable(search_control_test tests/search_control.cpp)
  target_link_libraries(search_control_test PRIVATE er2_engine)
  if(MINGW)
    target_link_options(search_control_test PRIVATE -static)
  endif()
  add_test(NAME live_search_controls COMMAND search_control_test "${CMAKE_CURRENT_BINARY_DIR}/test_results/search_controls")
  if(TARGET atm_er2_gui)
    add_test(NAME gui_button_and_worker COMMAND atm_er2_gui --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/gui_smoke.txt")
  endif()
endif()
